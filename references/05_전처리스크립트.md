
## 전처리 스크립트 (데이터 다운로드 후 실행)



```python

import pandas as pd

import numpy as np



# 1. 태양광 데이터 전처리

solar = pd.read_csv('data/solar/solar-energy-generation.csv')

solar['timestamp'] = pd.to_datetime(solar['timestamp'])

# 논문과 동일하게 해가 있는 시간만 (9:00~21:00)

solar = solar[solar['timestamp'].dt.hour.between(9, 20)]

solar['solar_power'] = solar['solar_power'] / 30.0  # 30MW로 정규화 (0~1)



# 2. 가격 데이터 전처리

da_prices = pd.read_csv('data/prices/day_ahead_prices.csv')

rt_prices = pd.read_csv('data/prices/real_time_prices.csv')

da_prices['timestamp'] = pd.to_datetime(da_prices['timestamp'])

rt_prices['timestamp'] = pd.to_datetime(rt_prices['timestamp'])



# 3. 기상 데이터 전처리

weather = pd.read_csv('data/weather/weather_data.csv')

weather['valid_time'] = pd.to_datetime(weather['valid_time'])



# 4. 통합

df = pd.merge(solar, da_prices, on='timestamp', how='left')

df = pd.merge(df, rt_prices, on='timestamp', how='left')

df = pd.merge(df, weather, left_on='timestamp', right_on='valid_time', how='left')



# 5. 학습/테스트 분리 (3개월 학습, 1개월 테스트)

train = df[df['timestamp'].dt.month.between(1, 3)]

test = df[df['timestamp'].dt.month == 4]



print(f"Train: {len(train)} rows, Test: {len(test)} rows")

```
